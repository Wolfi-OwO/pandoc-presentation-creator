  \ifnum\insertframenumber=1
    \relax % no footer on title slide
  \else
    \begin{beamercolorbox}[wd=\paperwidth,ht=3ex,dp=1ex,leftskip=0.5cm,rightskip=0.5cm]{footline}%
      {\usebeamerfont{title in head/foot}\footdate}%
      \hfill{\usebeamerfont{author in head/foot}\insertshortauthor}%
      \hfill{\usebeamerfont{title in head/foot}\insertframenumber\,\frameofframes\,\inserttotalframenumber}
    \end{beamercolorbox}%
  \fi}
