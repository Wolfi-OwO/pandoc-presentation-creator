% ==============================================================================
% Beamer Footer: Date | Middle | Page Number (16:9 Format)
% ==============================================================================

\setbeamercolor{footline}{bg=white, fg=black}
\setbeamerfont{footline}{size=\scriptsize}

\setbeamertemplate{footline}{
  \ifnum\insertframenumber>1
    \begin{beamercolorbox}[ht=2.5ex,dp=1.125ex,leftskip=.3cm,rightskip=.3cm]{footline}%
      {\usebeamerfont{footline}\usebeamercolor[fg]{footline}\insertdate}%
      \hfill%
      %
      \hfill%
      {\usebeamerfont{footline}\usebeamercolor[fg]{footline}\insertframenumber}
    \end{beamercolorbox}%
  \fi
}

% Remove navigation symbols
\setbeamertemplate{navigation symbols}{}
