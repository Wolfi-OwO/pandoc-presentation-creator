% Moderne Footer-Leiste mit Institution, Titel und Seitennummern

\setbeamertemplate{footline}{
  \ifnum\insertframenumber>1
    \leavevmode%
    \begin{beamercolorbox}[wd=\paperwidth,ht=2.5ex,dp=1.5ex,left,right]{footline}%
      \hspace*{0.5ex}%
      {\tiny \insertshortinstitute}%
      \hfill%
      {\tiny \insertshorttitle}%
      \hfill%
      {\tiny \insertframenumber{}/\inserttotalframenumber}%
      \hspace*{0.5ex}%
    \end{beamercolorbox}%
  \fi
}
