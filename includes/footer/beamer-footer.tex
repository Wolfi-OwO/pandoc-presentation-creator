% Professionelle Beamer Footer Template

% Moderne Footer-Leiste mit Institution, Titel und Seitennummern
% Improved visibility with better spacing and contrast

% Ensure footer line is visible
\setbeamercolor{footline}{bg=structure!5!white, fg=structure!50!black}
\setbeamerfont{footline}{size=\scriptsize}

\setbeamertemplate{footline}{
  \ifnum\insertframenumber>1
    \begin{beamercolorbox}[colsep=1.5pt]{upper separation line foot}
    \end{beamercolorbox}
    \begin{beamercolorbox}[ht=2.5ex,dp=1.125ex,leftskip=.3cm,rightskip=.3cm plus1fil]{footline}%
      {\usebeamerfont{footline}\usebeamercolor[fg]{footline}\insertshortinstitute}%
      \hfill%
      {\usebeamerfont{footline}\usebeamercolor[fg]{footline}\insertshorttitle}%
      \hfill%
      {\usebeamerfont{footline}\usebeamercolor[fg]{footline}\insertframenumber{}/\inserttotalframenumber}%
    \end{beamercolorbox}%
    \begin{beamercolorbox}[colsep=1.5pt]{lower separation line foot}
    \end{beamercolorbox}
  \fi
}

% Navigationssymbole entfernen für cleanes Design
\setbeamertemplate{navigation symbols}{}

% Better margin and spacing for footer
\setbeamertemplate{frametitle}[default][left]

% Ensure proper text sizing
\usefonttheme{structurebold}
